{{- /* Post cover. A post gets a hand-drawn cover when a partial named partials/covers/.html exists; otherwise it falls back to a procedural pattern keyed on the title. Safe band. Every cover draws on a 240x140 canvas with preserveAspectRatio "slice", which crops the canvas to a wide strip rather than fitting it. The strip is a different shape in each place a cover appears, and the tightest is the post detail page on a large screen. Measured bands, x is never cropped: post detail, 1032x220 y 44.4 to 95.6 <- tightest homepage featured, 840x192 y 42.6 to 97.4 list card, 406x144 y 27.4 to 112.6 So keep every cover inside y=46 to y=94. Wording especially: a caption that sits below y=94 loses its bottom half on the detail page. */ -}} {{- $p := .page -}} {{- $h := int (default 180 .height) -}} {{- $custom := "" -}} {{- with $p.File }}{{ $custom = printf "partials/covers/%s.html" .BaseFileName }}{{ end -}} {{- if and $custom (templates.Exists $custom) -}} {{ partial (strings.TrimPrefix "partials/" $custom) (dict "page" $p "height" $h) }} {{- else -}} {{- template "fallback-cover" (dict "page" $p "height" $h) -}} {{- end -}} {{- define "fallback-cover" -}} {{- $p := .page -}} {{- $h := int (default 180 .height) -}} {{- $bgs := slice "var(--pink)" "var(--mint)" "var(--lavender)" "var(--peach)" "var(--yellow-soft)" "var(--lime)" -}} {{- $seed := len $p.Title -}} {{- $bg := index $bgs (mod $seed (len $bgs)) -}} {{- $variant := mod $seed 3 -}} {{- $label := "" -}}{{- with $p.Params.tags }}{{ $label = index . 0 }}{{ end -}} {{- end -}}
