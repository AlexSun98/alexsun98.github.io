{{- /* Post cover. A post gets a hand-drawn cover when a partial named partials/covers/.html exists; otherwise it falls back to a procedural pattern keyed on the title. */ -}} {{- $p := .page -}} {{- $h := int (default 180 .height) -}} {{- $custom := "" -}} {{- with $p.File }}{{ $custom = printf "partials/covers/%s.html" .BaseFileName }}{{ end -}} {{- if and $custom (templates.Exists $custom) -}} {{ partial (strings.TrimPrefix "partials/" $custom) (dict "page" $p "height" $h) }} {{- else -}} {{- template "fallback-cover" (dict "page" $p "height" $h) -}} {{- end -}} {{- define "fallback-cover" -}} {{- $p := .page -}} {{- $h := int (default 180 .height) -}} {{- $bgs := slice "var(--pink)" "var(--mint)" "var(--lavender)" "var(--peach)" "var(--yellow-soft)" "var(--lime)" -}} {{- $seed := len $p.Title -}} {{- $bg := index $bgs (mod $seed (len $bgs)) -}} {{- $variant := mod $seed 3 -}} {{- $label := "" -}}{{- with $p.Params.tags }}{{ $label = index . 0 }}{{ end -}} {{- end -}}
